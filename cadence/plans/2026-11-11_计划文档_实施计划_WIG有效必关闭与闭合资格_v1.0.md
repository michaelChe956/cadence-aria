  requires_closure(finding, scope) =
      finding.is_actionable()                         // 硬化后的 disposition（含缺省视作 actionable）

          severity ∈ {Blocking, MustFix}
          || (scope 为 WorkItemPlan

  auto_repair_permitted(finding, scope) =
